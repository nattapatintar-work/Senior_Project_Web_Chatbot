# Python dependencies for the Ingredient-to-Recipe Chatbot (Person 2 side)
#
# Install everything with:
#     pip install -r requirements.txt
#
# ">=" means "this version or newer". Loose pins are fine for a student
# project. Before the final demo, run `pip freeze > requirements.lock.txt` to
# capture the exact versions that are known to work.

# --- Web / LINE ------------------------------------------------------------
flask>=3.0.0             # web server that receives the LINE webhook (Week 2)
line-bot-sdk>=3.11.0     # official LINE client: verify signatures, send replies
python-dotenv>=1.0.0     # load secrets from a .env file instead of hardcoding
                         # them — a leaked channel token lets anyone post as
                         # your bot, so .env must never be committed to git

# --- Web chatbot backend (api/app.py) ----------------------------------------
fastapi>=0.115.0         # REST framework for the six web endpoints
uvicorn>=0.30.0          # ASGI server:  uvicorn api.app:app --port 8000
python-multipart>=0.0.9  # required by FastAPI to parse the /detect image upload
slowapi>=0.1.9           # per-IP rate limiting
httpx>=0.27.0            # FastAPI TestClient (tests/test_web_api.py)

# --- NLP (Week 5) ----------------------------------------------------------
pythainlp>=5.0.0         # Thai word segmentation; Thai has no spaces between
                         # words, so text must be tokenized before matching
rapidfuzz>=3.0.0         # fuzzy matching for typos in ingredient names

# --- Recommender (Week 7) --------------------------------------------------
scikit-learn>=1.4.0      # TfidfVectorizer + cosine_similarity, both built in

# --- Chat system (Week 8) ---------------------------------------------------
pyyaml>=6.0               # reads thresholds.yaml (LLM routing config) from Person 1

# --- Computer vision (Week 8/9) ---------------------------------------------
ultralytics>=8.4.0        # loads and runs Person 1's delivered YOLO model
                          # (models/best_phase1_n_ceiling1000_img640.pt). Pulls
                          # in torch as its own dependency automatically; no
                          # separate torch pin needed here.

# --- Testing ---------------------------------------------------------------
pytest>=8.0.0            # runs tests/test_contract.py
